CREATE TABLE IF NOT EXISTS public.clientes
(
    id_cliente integer NOT NULL DEFAULT nextval('clientes_id_cliente_seq'::regclass),
    cpf_cnpj character varying(20) COLLATE pg_catalog."default" NOT NULL,
    nome character varying(100) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT clientes_pkey PRIMARY KEY (id_cliente)
)

CREATE TABLE IF NOT EXISTS public.despesas
(
    id_despesa integer NOT NULL DEFAULT nextval('despesas_id_despesa_seq'::regclass),
    numero_os_projeto integer NOT NULL,
    data_despesa date NOT NULL,
    observacao text COLLATE pg_catalog."default",
    mao_de_obra numeric(12,2) DEFAULT 0,
    alimentacao numeric(12,2) DEFAULT 0,
    hospedagem numeric(12,2) DEFAULT 0,
    viagem numeric(12,2) DEFAULT 0,
    seguranca_trabalho numeric(12,2) DEFAULT 0,
    material numeric(12,2) DEFAULT 0,
    equipamento numeric(12,2) DEFAULT 0,
    andaime numeric(12,2) DEFAULT 0,
    documentacao numeric(12,2) DEFAULT 0,
    outros numeric(12,2) DEFAULT 0,
    total numeric(12,2) GENERATED ALWAYS AS ((((((((((mao_de_obra + alimentacao) + hospedagem) + viagem) + seguranca_trabalho) + material) + equipamento) + andaime) + documentacao) + outros)) STORED,
    CONSTRAINT despesas_pkey PRIMARY KEY (id_despesa),
    CONSTRAINT despesas_numero_os_projeto_fkey FOREIGN KEY (numero_os_projeto)
        REFERENCES public.projetos (numero_os) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.itens_lista_materiais
(
    id integer NOT NULL DEFAULT nextval('itens_lista_materiais_id_seq'::regclass),
    id_lista integer NOT NULL,
    produto_id integer NOT NULL,
    nome_produto character varying(100) COLLATE pg_catalog."default",
    quantidade numeric(10,2) NOT NULL,
    unidade character varying(20) COLLATE pg_catalog."default" NOT NULL,
    preco_unitario numeric(10,2) NOT NULL,
    observacao text COLLATE pg_catalog."default",
    CONSTRAINT itens_lista_materiais_pkey PRIMARY KEY (id),
    CONSTRAINT itens_lista_materiais_id_lista_fkey FOREIGN KEY (id_lista)
        REFERENCES public.lista_materiais (id_lista) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT itens_lista_materiais_produto_id_fkey FOREIGN KEY (produto_id)
        REFERENCES public.produtos (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.lista_materiais
(
    id_lista integer NOT NULL DEFAULT nextval('lista_materiais_id_lista_seq'::regclass),
    os_referencia character varying(50) COLLATE pg_catalog."default",
    responsavel character varying(100) COLLATE pg_catalog."default" NOT NULL,
    observacao text COLLATE pg_catalog."default",
    data_criacao timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT lista_materiais_pkey PRIMARY KEY (id_lista)
)

CREATE TABLE IF NOT EXISTS public.orcamentos

CREATE TABLE IF NOT EXISTS public.produtos
(
    id integer NOT NULL DEFAULT nextval('produtos_id_seq'::regclass),
    nome character varying(100) COLLATE pg_catalog."default" NOT NULL,
    descricao text COLLATE pg_catalog."default",
    quantidade integer NOT NULL DEFAULT 0,
    unidade character varying(20) COLLATE pg_catalog."default",
    preco_unitario numeric(10,2),
    estoque_minimo integer DEFAULT 0,
    criado_em timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT produtos_pkey PRIMARY KEY (id)
)

CREATE TABLE IF NOT EXISTS public.projetos
(
    numero_os integer NOT NULL DEFAULT nextval('seq_numero_os'::regclass),
    tipo character varying(50) COLLATE pg_catalog."default" NOT NULL,
    id_cliente integer NOT NULL,
    cliente_nome character varying(100) COLLATE pg_catalog."default" NOT NULL,
    cliente_cpf_cnpj character varying(20) COLLATE pg_catalog."default" NOT NULL,
    data_os date NOT NULL,
    numero_proposta character varying(20) COLLATE pg_catalog."default",
    valor_servico numeric(12,2),
    valor_material numeric(12,2),
    total numeric(12,2),
    endereco_obra character varying(200) COLLATE pg_catalog."default",
    cidade_obra character varying(50) COLLATE pg_catalog."default",
    estado_obra character(2) COLLATE pg_catalog."default",
    contato character varying(100) COLLATE pg_catalog."default",
    nome_responsavel character varying(100) COLLATE pg_catalog."default",
    status character varying(30) COLLATE pg_catalog."default",
    CONSTRAINT projetos_pkey PRIMARY KEY (numero_os),
    CONSTRAINT projetos_id_cliente_fkey FOREIGN KEY (id_cliente)
        REFERENCES public.clientes (id_cliente) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.propostas
(
    id integer NOT NULL DEFAULT nextval('propostas_id_seq'::regclass),
    titulo character varying(255) COLLATE pg_catalog."default" NOT NULL,
    descricao text COLLATE pg_catalog."default",
    valor numeric(10,2) NOT NULL,
    status character varying(50) COLLATE pg_catalog."default" NOT NULL DEFAULT 'pendente'::character varying,
    criado_em timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT propostas_pkey PRIMARY KEY (id)
)

CREATE TABLE IF NOT EXISTS public.receitas
(
    id_receita integer NOT NULL DEFAULT nextval('receitas_id_receita_seq'::regclass),
    numero_os_projeto integer NOT NULL,
    data_receita date NOT NULL,
    nf character varying(20) COLLATE pg_catalog."default",
    cliente character varying(100) COLLATE pg_catalog."default" NOT NULL,
    valor_servico numeric(12,2),
    valor_material numeric(12,2),
    imposto numeric(5,2),
    icms numeric(5,2),
    valor_liquido numeric(12,2) NOT NULL,
    CONSTRAINT receitas_pkey PRIMARY KEY (id_receita),
    CONSTRAINT receitas_numero_os_projeto_fkey FOREIGN KEY (numero_os_projeto)
        REFERENCES public.projetos (numero_os) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

CREATE TABLE IF NOT EXISTS public.usuarios
(
    id integer NOT NULL DEFAULT nextval('usuarios_id_seq'::regclass),
    nome character varying(100) COLLATE pg_catalog."default" NOT NULL,
    login character varying(50) COLLATE pg_catalog."default" NOT NULL,
    senha character varying(255) COLLATE pg_catalog."default" NOT NULL,
    grupo character varying(20) COLLATE pg_catalog."default" NOT NULL,
    ativo boolean NOT NULL DEFAULT true,
    CONSTRAINT usuarios_pkey PRIMARY KEY (id),
    CONSTRAINT usuarios_login_key UNIQUE (login),
    CONSTRAINT usuarios_grupo_check CHECK (grupo::text = ANY (ARRAY['estoquista'::character varying, 'vendedor'::character varying, 'engenheiro'::character varying, 'financeiro'::character varying, 'gerencia'::character varying, 'ti'::character varying]::text[]))
)

CREATE TABLE IF NOT EXISTS auditoria (
    id SERIAL PRIMARY KEY,
    nome_usuario VARCHAR(100) NOT NULL,
    acao VARCHAR(20) NOT NULL,             -- Ex: INSERT, UPDATE, DELETE, LOGIN, LOGOUT, RELATORIO_GERADO
    modulo VARCHAR(50) NOT NULL,           -- Ex: usuarios, relatorios, login, projetos
    descricao TEXT,                        -- Ex: "Usuário alterou grupo de 'joao' para 'financeiro'"
    data_hora TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
